O FF00
MVI B,1
MOV C,B

MOV A,C
ORA A
JPO FF2A   ; ODD
ANI 5
MOV D,A
JPO FF16   ; OUT_FIZZ 
MOV A,C
ORA A
JZ FF16    ; OUT_FIZZ 
JMP FF21   ; OUT_BUZZ 

LXI H,FF7B ; FIZZ 
CALL 0014
MOV A,D
ORA A
JPO FF2E   ; NEXT 

LXI H,FF80 ; BUZZ 
CALL 0014
JMP FF2E   ; NEXT 

MOV A,B 
CALL FF45  ; OUT_NUM 

CALL 0018  ; PUTSP

INR B
INR C
MOV A,C
CPI 0F     ; 10?
JNZ FF3B   ; NEXT2 
MVI C,0

MOV A,B
CPI 65
JNZ FF03   ; LOOP 
CALL 0469  ; PUTCRLF

RET  

PUSH PSW 
PUSH B 
PUSH D 
MVI B,0 

CPI 64
JC FF55    ; TEN 
SUI 64
INR B
JMP FF4A   ; HAND 

MVI C,0

CPI 0A     ; >10?
JC FF62    ; ONE
SUI 0A     ; -10
INR C
JMP FF57   ; TENS 

MOV D,A
MOV A,B
ORA A
JZ FF6B    ; SKIP10 
ADI 30     ; TO ASCII
RST 2      ; PUTCH

MOV A,C
ORA A
JZ FF73    ; SKIP1 
ADI 30     ; TO ASCII
RST 2      ; PUTCH

MOV A,D
ADI 30     ; TO ASCII
RST 2      ; PUTCH
POP D
POP B
POP PSW
RET
